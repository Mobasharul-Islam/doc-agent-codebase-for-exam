"""HITL -- human-in-the-loop review queue"""
from __future__ import annotations

from ..contracts import *  # noqa
from . import hitl_store

def escalate(reason: str, context: dict) -> ToolResult:
    """Queue for human review. Returns status 'pending': the caller must NOT act until
    hitl_store.decision(item_id) == 'approved' (the action is blocked, not skipped)."""
    item_id = hitl_store.enqueue({"reason": reason, "context": context})
    return ToolResult(ok=True, payload={"item_id": item_id, "status": "pending", "blocked": True})

def review_queue():
    """Return pending items for the reviewer UI."""
    return hitl_store.pending()
